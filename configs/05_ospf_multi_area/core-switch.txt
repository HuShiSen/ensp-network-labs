# =============================================================================
# 05 OSPF 多区域：骨干 Area 0 + 业务 Area 10（本机做 ABR）
# -----------------------------------------------------------------------------
# 设备：S5700（LSW-CORE-A，ABR：同时属于 Area 0 与 Area 10）
# 拓扑假设：
#   Area 0（骨干）：核心互联段 10.10.0.0/30、管理 VLAN999、与 AR-EDGE 的互联
#   Area 10（业务）：VLAN 10/20/30 三个业务网段，由本机注入骨干
# 地址：router-id 用 Loopback0（10.255.0.1）
# 下发顺序：01~04 完成、二层与地址就绪之后
# 验收：
#   display ospf peer brief      → 所有邻居 State = Full（Area 0 与 Area 10 各自建立）
#   display ospf brief           → 本机角色含 ABR
#   display ospf lsdb            → Area 0 中有 Type-3（Summary）LSA
#   display ip routing-table protocol ospf
#                                → 对端区域网段存在，下一跳正确
# 常用调优：
#   1) abr-summary 汇总业务网段，减少 LSA 数量
#   2) 接口下 ospf cost 调整选路（默认按带宽，Eth-Trunk 与单口需手工干预）
#   3) area 10 设为 stub/nssa 时，注意 ASBR 与外部路由的引入方式
# =============================================================================

sysname LSW-CORE-A

# ---- Loopback 作为 router-id（必须显式指定，避免用接口地址漂移）----
interface LoopBack0
 description router-id
 ip address 10.255.0.1 255.255.255.255

# ---- 三层接口（业务网关由 07_vrrp 提供虚拟 IP，这里配物理地址）----
interface Vlanif10
 description OFFICE-gw
 ip address 192.168.10.1 255.255.255.0
interface Vlanif20
 description VIDEO-gw
 ip address 192.168.20.1 255.255.255.0
interface Vlanif30
 description COUNTER-gw
 ip address 192.168.30.1 255.255.255.0

# ---- 与 AR-EDGE 的互联（示例：物理口直连，/30）----
interface GigabitEthernet0/0/23
 description to-AR-EDGE-GE0/0/1
 undo portswitch                       # 三层口（部分型号需先转三层）
 ip address 10.10.0.1 255.255.255.252

# ---- OSPF 进程：Area 0 + Area 10 ----
ospf 1 router-id 10.255.0.1
 area 0.0.0.0
  network 10.10.0.0 0.0.0.3            # 与 AR-EDGE 的互联
  network 192.168.99.0 0.0.0.255       # 管理 VLAN
 area 0.0.0.10
  network 192.168.10.0 0.0.0.255
  network 192.168.20.0 0.0.0.255
  network 192.168.30.0 0.0.0.255
  abr-summary 192.168.0.0 255.255.0.0  # 业务网段汇总后注入骨干，减少 LSA
  # 可选：area 0.0.0.10 设为 stub（区域内不能有 ASBR）
  # stub

# interface GigabitEthernet0/0/23
#  ospf cost 10
