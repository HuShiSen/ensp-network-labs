sysname <DEVICE-NAME>

clock timezone CST add 08:00:00
clock datetime <YYYY-MM-DD> <HH:MM:SS>

# ---- 接口与区域（区域名：trust / untrust / dmz / ha，按需增减）----
interface GigabitEthernet1/0/0
 description to-untrust
 ip address <IP> <MASK>
 arp-proxy enable                 # 需要做目的 NAT 发布时必开

interface GigabitEthernet1/0/1
 description to-trust
 ip address <IP> <MASK>

firewall zone untrust
 add interface GigabitEthernet1/0/0
firewall zone trust
 add interface GigabitEthernet1/0/1

# ---- 默认策略：默认拒绝，只放通业务需要的方向 ----
security-policy
 rule name trust-to-untrust
  source-zone trust
  destination-zone untrust
  action permit

# ---- 日志 ----
