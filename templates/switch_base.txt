# 交换机基础配置模板（S5700/S3700 通用，落地前按规划的 VLAN/网段替换）
# 用途：每台交换机开局先跑这一段，避免"每台设备各写一套"
# =============================================================================

sysname <DEVICE-NAME>

# ---- 时区与时间（日志时间戳准确，排错才有意义）----
clock timezone CST add 08:00:00
clock datetime <YYYY-MM-DD> <HH:MM:SS>

# ---- 日志与告警 ----
info-center loghost 192.168.99.100          # 可选：接日志服务器
info-center timestamp loghost date
info-center source default channel 2 log level warning

# ---- 生成树：全网统一 RSTP ----
stp mode rstp
# stp instance 0 root primary               # 核心 A
# stp instance 0 root secondary             # 核心 B

# ---- 管理 VLAN 与地址 ----
vlan batch 10 20 30 999
interface Vlanif999
 description mgmt
 ip address 192.168.99.<N> 255.255.255.0
